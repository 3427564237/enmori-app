import 'dart:convert';

import 'package:enmori/core/network/backend_api_client.dart';
import 'package:enmori/core/result/result.dart';

import 'package:http/http.dart' as http;

const key = '11111111-1111-4111-8111-111111111111';
const fields = PersonaFields(
  displayName: 'Guide',
  tone: PersonaTone.neutral,
  responseLength: PersonaResponseLength.balanced,
  formatStyle: PersonaFormatStyle.plain,
);
final row = <String, Object?>{
  'personaId': 'a' * 64,
  'displayName': 'Guide',
  'tone': 'neutral',
  'responseLength': 'balanced',
  'formatStyle': 'plain',
  'revision': 1,
  'createdAt': '2026-10-09T00:00:00.000Z',
  'updatedAt': '2026-10-09T00:00:00.000Z',
};
Map<String, Object?> get saved => {'persona': row, 'stateRevision': 1};
http.StreamedResponse reply(int status, Object body) => http.StreamedResponse(
  Stream.value(utf8.encode(jsonEncode(body))),
  status,
  headers: {'content-type': 'application/json'},
);

void main() {
  for (final boundary in ['initial-read', 'create-token']) {
    test('original assumption with controlled $boundary timeout', () async {
      final transport = _Transport(delayedBody: false)
        ..holdInitialRead = boundary == 'initial-read';
      var tokens = 0;
      final client = BackendApiClient(Uri.parse('https://example.test'), () async {
        tokens++;
        if (boundary == 'create-token' && tokens == 2) {
          await Future<void>.delayed(const Duration(milliseconds: 100));
        }
        return const Success('synthetic');
      }, httpClient: transport, operationTimeout: const Duration(milliseconds: 40));
      final session = PersonaSession(BackendPersonaRepository(client), newKey: () => key);
      addTearDown(() { session.dispose(); client.close(); });
      void capture(String stage) => print(jsonEncode({
        'boundary': boundary, 'stage': stage, 'current': session.state.isCurrent,
        'canCreate': session.state.canStartNewCreate,
        'pending': session.state.pendingCreate?.status.name,
        'failure': session.state.failure?.kind.name,
        'gets': transport.gets, 'posts': transport.posts.length, 'tokenCalls': tokens,
      }));
      await session.load();
      capture('after-load');
      expect(await session.create(fields), isFalse);
      capture('after-create');
      // Preserve the original failing assertion to reproduce the same symptom.
      expect(session.state.pendingCreate!.status, PersonaCreateStatus.uncertain);
    });
  }
}
final class _Transport extends http.BaseClient {
  _Transport({required this.delayedBody});
  final bool delayedBody;
  bool immediate = false;
  bool holdInitialRead = false;
  bool committed = false;
  int gets = 0;
  final posts = <http.Request>[];
  final headers = Completer<http.StreamedResponse>();
  final body = StreamController<List<int>>();
  @override
  Future<http.StreamedResponse> send(http.BaseRequest request) async {
    if (request.method == 'GET') {
      gets++;
      if (holdInitialRead && gets == 1) { await Future<void>.delayed(const Duration(milliseconds: 100)); }
      return reply(200, {
        'personas': {
          'items': committed ? [row] : [],
          'stateRevision': committed ? 1 : 0,
        },
      });
    }
    posts.add(request as http.Request);
    if (immediate) {
      committed = true;
      return reply(201, saved);
    }
    if (posts.length == 1) {
      return delayedBody
          ? http.StreamedResponse(
              body.stream,
              201,
              headers: {'content-type': 'application/json'},
            )
          : headers.future;
    }
    return reply(412, {
      'error': {
        'code': 'PERSONA_REVISION_PRECONDITION_FAILED',
        'message': 'Your Personas changed. Reload them before trying again.',
        'requestId': 'synthetic-request',
      },
    });
  }

  void release() {
    committed = true;
    if (delayedBody) {
      body.add(utf8.encode(jsonEncode(saved)));
      unawaited(body.close());
    } else {
      headers.complete(reply(201, saved));
    }
  }
}
